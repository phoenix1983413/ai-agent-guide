第 10 章

可观测性与调试

Tracing / 结构化日志 / 回放复现 / 执行可视化

10.1 追踪（Tracing）：Agent 执行的完整调用链

Agent 的一次执行可能涉及多次 LLM 调用、工具调用、记忆检索。Tracing 把这些串成一条可追溯的链。核心字段：

  • trace_id：整个任务的唯一 ID。
  • span_id：每个步骤（LLM 调用、工具调用）的唯一 ID。
  • parent_span_id：父子关系，形成调用树。
  • attributes：模型名称、温度、输入输出 token、工具名、参数、返回值、耗时。

推荐用 OpenTelemetry（OTel）作为标准，后端选 Jaeger / Grafana Tempo / 阿里云链路追踪。不要自己发明格式——行业标准能直接接现有监控栈。

10.2 日志设计：结构化日志 vs 自然语言摘要

日志有两种用途：机器检索和人工阅读。好的设计是「结构化为主，自然语言为辅」：

  • 结构化日志（JSON）：用于检索、告警、聚合。每条日志包含 timestamp、level、trace_id、span_id、agent_id、event、data。
  • 自然语言摘要：每次任务结束生成一段人话总结，如「Agent 调用 search 3 次、calc 1 次，耗时 4.2s，最终回答基于前两条搜索结果」。方便人工快速 review。

10.3 回放与复现：把一次执行变成可重复的测试用例

Agent 的行为有随机性（temperature > 0），但调试时需要复现。回放（Replay）机制：

  1. 录制：生产环境把 trace + 所有 LLM 输入输出 + 工具返回值持久化。
  2. 脱敏：回放前抹掉 API key、用户 PII。
  3. 复现：用录制的输入顺序驱动 Agent，断言每一步的输出与录制一致。
  4. 差分：prompt 变更后，自动重跑历史 trace，检测行为漂移。

10.4 可视化：执行图、时间线、成本热力图

可视化是调试 Agent 的最高效手段。三种视图：

  • 执行图（Call Graph）：节点是 LLM 调用或工具调用，边是数据流。一眼看出哪个分支走了弯路。
  • 时间线（Timeline）：横轴是时间，每个 span 是一条色块。找出耗时异常的步骤。
  • 成本热力图：按任务类型 / 模型 / 工具统计 token 消耗和费用。定位「烧钱大户」。

本章产出：OpenTelemetry Agent 追踪模板（双语言）

下面是一个最小但完整的 OTel 集成模板，把 Agent 的一次执行包装成 trace + span。

Python 版本

"""
OpenTelemetry Agent 追踪模板 —— Python
依赖：opentelemetry-api, opentelemetry-sdk, opentelemetry-exporter-otlp
"""
from __future__ import annotations
import time
from dataclasses import dataclass, field
from typing import Any
from opentelemetry import trace
from opentelemetry.sdk.trace import TracerProvider
from opentelemetry.sdk.trace.export import BatchSpanProcessor
from opentelemetry.exporter.otlp.proto.grpc.trace_exporter import OTLPSpanExporter

# ---- OTel 初始化 ----
provider = TracerProvider()
processor = BatchSpanProcessor(OTLPSpanExporter(endpoint="http://localhost:4317", insecure=True))
provider.add_span_processor(processor)
trace.set_tracer_provider(provider)
tracer = trace.get_tracer("agent")

@dataclass
class SpanRecord:
    name: str
    attributes: dict[str, Any] = field(default_factory=dict)
    start_time: float = field(default_factory=time.time)
    end_time: float | None = None

class TracedAgent:
    """包装 Agent 执行，自动记录 trace。"""
    def __init__(self, agent_fn):
        self.agent_fn = agent_fn
        self.spans: list[SpanRecord] = []

    def run(self, user_input: str) -> dict:
        with tracer.start_as_current_span("agent.run") as span:
            span.set_attribute("input", user_input[:200])
            # 记录工具调用
            tools_used = []
            def tool_hook(name: str, args: dict, result: Any, latency_ms: float):
                tools_used.append(name)
                span.add_event("tool_call", {
                    "tool": name,
                    "args": str(args)[:200],
                    "result": str(result)[:200],
                    "latency_ms": latency_ms,
                })
            # 执行 Agent（注入 hook）
            output = self.agent_fn(user_input, tool_hook=tool_hook)
            span.set_attribute("tools_used", tools_used)
            span.set_attribute("output_length", len(output.get("text", "")))
            return output

# 使用示例
def my_agent(input: str, tool_hook) -> dict:
    # 模拟工具调用
    tool_hook("search", {"q": input}, "3 results", 120)
    return {"text": f"回答：{input}", "tokens": 50}

if __name__ == "__main__":
    agent = TracedAgent(my_agent)
    result = agent.run("如何学习 AI Agent？")
    print("结果：", result["text"])

TypeScript 版本

/**
OpenTelemetry Agent 追踪模板 —— TypeScript
依赖：@opentelemetry/api, @opentelemetry/sdk-trace-node, @opentelemetry/exporter-trace-otlp-grpc
*/
import { trace, Span, TraceFlags } from "@opentelemetry/api";
import { NodeTracerProvider } from "@opentelemetry/sdk-trace-node";
import { OTLPSpanExporter } from "@opentelemetry/exporter-trace-otlp-grpc";

// ---- OTel 初始化 ----
const provider = new NodeTracerProvider();
const exporter = new OTLPSpanExporter({ url: "http://localhost:4317" });
provider.addSpanProcessor(new { forceFlush(): void {} } as any); // 简化：生产环境用 BatchSpanProcessor
provider.register();
const tracer = trace.getTracer("agent");

interface SpanRecord {
  name: string;
  attributes: Record;
  startTime: number;
}

class TracedAgent {
  private spans: SpanRecord[] = [];

  constructor(private agentFn: (input: string, hook: ToolHook) => Promise>) {}

  async run(userInput: string): Promise> {
    return tracer.startActiveSpan("agent.run", async (span) => {
      span.setAttribute("input", userInput.slice(0, 200));
      const toolsUsed: string[] = [];
      const toolHook: ToolHook = (name, args, result, latencyMs) => {
        toolsUsed.push(name);
        span.addEvent("tool_call", { tool: name, args: JSON.stringify(args).slice(0, 200), result: String(result).slice(0, 200), latencyMs });
      };
      const output = await this.agentFn(userInput, toolHook);
      span.setAttribute("tools_used", toolsUsed);
      span.setAttribute("output_length", output.text?.length ?? 0);
      span.end();
      return output;
    });
  }
}

type ToolHook = (name: string, args: Record, result: any, latencyMs: number) => void;

// 使用示例
async function myAgent(input: string, hook: ToolHook): Promise> {
  const start = Date.now();
  const result = "3 results";
  hook("search", { q: input }, result, Date.now() - start);
  return { text: `回答：${input}`, tokens: 50 };
}

(async () => {
  const agent = new TracedAgent(myAgent);
  const result = await agent.run("如何学习 AI Agent？");
  console.log("结果：", result.text);
})();
