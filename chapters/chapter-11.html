第 11 章

安全、合规与治理

权限网关 / PII 脱敏 / 内容安全 / 审计不可抵赖 / 合规上线

Agent 和传统软件最大的区别是：它会自己做决定、自己调工具、自己写内容。传统应用把风险关在代码逻辑里，Agent 把风险摊到了每一次模型推理上。这一章把安全从「上线前评审」变成「运行时每一层都在编译」的工程体系。

11.1 权限最小化：工具调用的能力边界

Agent 能做什么，不取决于 prompt 怎么写，而取决于运行时它手里有什么工具、每个工具以什么身份执行。核心原则：

  • 默认拒绝：工具注册表走白名单制，不在表里的调用直接拒绝，而不是「先执行再判断」。
  • 能力拆分：「邮件」拆成「读邮件」「写草稿」「发邮件」三个工具，Agent 通常只拿到前两个，发送必须人确认。
  • 工具级幂等与配额：每个工具带速率限制（如「查询订单」每分钟 30 次）和金额上限（如「退款」单笔 ≤ 200 元）。
  • 会话级授权：权限绑定到 user_id + session_id，防止 Agent 拿到别人的上下文后越权。

下面的工具权限网关把上述规则集中到一个可测试的组件里：每次工具调用先过 check()，通过才放行执行，执行后无论成败都记审计日志。

Python 版本

"""
工具权限网关 —— Python
职责：白名单 / 危险操作确认 / 配额 / 审计日志，落在每次工具调用前后。
"""
from __future__ import annotations
import time, json, hashlib
from dataclasses import dataclass

@dataclass
class ToolPolicy:
    name: str
    danger_level: int = 0          # 0 只读 / 1 写 / 2 不可逆（删除、支付）
    rate_per_min: int = 60
    amount_limit: float | None = None

class AuditLog:
    """不可抵赖审计：追加写 + 哈希链，改一条就会断链。"""
    def __init__(self):
        self.chain: list[dict] = []
        self._prev_hash = "GENESIS"

    def append(self, event: dict) -> dict:
        payload = json.dumps(event, ensure_ascii=False, sort_keys=True) + self._prev_hash
        self._prev_hash = hashlib.sha256(payload.encode()).hexdigest()[:16]
        record = {**event, "ts": time.time(), "hash": self._prev_hash}
        self.chain.append(record)
        return record

class ToolGateway:
    def __init__(self, policies: dict[str, ToolPolicy], audit: AuditLog):
        self.policies = policies
        self.audit = audit
        self._calls: dict[str, list[float]] = {}
        self._approved: set[str] = set()   # 人工已确认的危险调用 token

    def check(self, tool: str, args: dict, session: str) -> tuple[bool, str]:
        if tool not in self.policies:
            return False, f"工具 {tool} 未注册，默认拒绝"
        policy = self.policies[tool]
        # 1. 危险级别 2 → 必须有人的确认 token
        key = f"{session}:{tool}:{args.get('id', '')}"
        if policy.danger_level >= 2 and key not in self._approved:
            return False, "不可逆操作，需人工确认后携带 approval_token 重试"
        # 2. 金额上限
        if policy.amount_limit is not None:
            amount = float(args.get("amount", 0))
            if amount > policy.amount_limit:
                return False, f"金额 {amount} 超过单笔上限 {policy.amount_limit}"
        # 3. 配额（滑动窗口）
        now = time.time()
        window = [t for t in self._calls.get(tool, []) if now - t < 60]
        if len(window) >= policy.rate_per_min:
            return False, "超出速率限制"
        window.append(now)
        self._calls[tool] = window
        return True, "ok"

    def approve(self, session: str, tool: str, target_id: str) -> str:
        token = f"{session}:{tool}:{target_id}"
        self._approved.add(token)
        return token

    def invoke(self, tool: str, args: dict, session: str, fn) -> dict:
        ok, reason = self.check(tool, args, session)
        self.audit.append({"event": "tool_call", "tool": tool,
                           "args": {k: str(v)[:64] for k, v in args.items()},
                           "session": session, "allowed": ok, "reason": reason})
        if not ok:
            return {"ok": False, "error": reason}
        try:
            result = fn(**args)
            self.audit.append({"event": "tool_result", "tool": tool, "session": session, "ok": True})
            return {"ok": True, "result": result}
        except Exception as e:
            self.audit.append({"event": "tool_error", "tool": tool,
                               "session": session, "ok": False, "error": str(e)[:200]})
            return {"ok": False, "error": str(e)}

# ---- 使用示例 ----
policies = {
    "search_orders": ToolPolicy("search_orders", danger_level=0, rate_per_min=30),
    "draft_reply":    ToolPolicy("draft_reply",    danger_level=1, rate_per_min=10),
    "issue_refund":   ToolPolicy("issue_refund",   danger_level=2, rate_per_min=5, amount_limit=200.0),
    "delete_ticket":  ToolPolicy("delete_ticket",  danger_level=2, rate_per_min=2),
}
audit = AuditLog()
gateway = ToolGateway(policies, audit)

# Agent 直接发起 500 元退款 → 未过人工确认，被拦
print(gateway.invoke("issue_refund", {"amount": 500, "id": "T-9"}, "sess-1", lambda **kw: None))
# 人工确认后重试 120 元 → 在限额内，放行
gateway.approve("sess-1", "issue_refund", "T-9")
print(gateway.invoke("issue_refund", {"amount": 120, "id": "T-9"}, "sess-1", lambda **kw: "REFUNDED"))

TypeScript 版本

/**
 * 工具权限网关 —— TypeScript
 */
type DangerLevel = 0 | 1 | 2;

interface ToolPolicy {
  dangerLevel?: DangerLevel;        // 0 只读 / 1 写 / 2 不可逆
  ratePerMin?: number;
  amountLimit?: number | null;
}

type CheckResult = { ok: true } | { ok: false; reason: string };

export class AuditLog {
  private chain: Array<Record<string, unknown>> = [];
  private prevHash = "GENESIS";

  append(event: Record<string, unknown>) {
    const payload = JSON.stringify(event) + this.prevHash;
    let h = 0;                                   // 生产环境用 crypto.subtle.digest
    for (const ch of payload) h = (h * 31 + ch.charCodeAt(0)) | 0;
    this.prevHash = Math.abs(h).toString(16).padStart(8, "0");
    const record = { ...event, ts: Date.now(), hash: this.prevHash };
    this.chain.push(record);
    return record;
  }
}

export class ToolGateway {
  private calls = new Map<string, number[]>();
  private approved = new Set<string>();

  constructor(
    private policies: Map<string, ToolPolicy>,
    private audit: AuditLog,
  ) {}

  check(tool: string, args: Record<string, unknown>, session: string): CheckResult {
    const policy = this.policies.get(tool);
    if (!policy) return { ok: false, reason: `工具 ${tool} 未注册，默认拒绝` };
    if ((policy.dangerLevel ?? 0) >= 2
        && !this.approved.has(`${session}:${tool}:${String(args.id ?? "")}`)) {
      return { ok: false, reason: "不可逆操作，需人工确认后携带 approval_token 重试" };
    }
    if (policy.amountLimit != null && Number(args.amount ?? 0) > policy.amountLimit) {
      return { ok: false, reason: `金额超过单笔上限 ${policy.amountLimit}` };
    }
    const now = Date.now();
    const window = (this.calls.get(tool) ?? []).filter((t) => now - t < 60_000);
    if (window.length >= (policy.ratePerMin ?? 60)) {
      return { ok: false, reason: "超出速率限制" };
    }
    window.push(now);
    this.calls.set(tool, window);
    return { ok: true };
  }

  approve(session: string, tool: string, targetId: string): string {
    const token = `${session}:${tool}:${targetId}`;
    this.approved.add(token);
    return token;
  }

  async invoke<A extends Record<string, unknown>, R>(
    tool: string, args: A, session: string, fn: (a: A) => Promise<R>,
  ): Promise<{ ok: true; result: R } | { ok: false; error: string }> {
    const res = this.check(tool, args, session);
    this.audit.append({ event: "tool_call", tool, session, ok: res.ok,
                        reason: res.ok ? undefined : res.reason });
    if (!res.ok) return { ok: false, error: res.reason };
    try {
      const result = await fn(args);
      this.audit.append({ event: "tool_result", tool, session, ok: true });
      return { ok: true, result };
    } catch (e) {
      this.audit.append({ event: "tool_error", tool, session, ok: false,
                          reason: String(e).slice(0, 200) });
      return { ok: false, error: String(e) };
    }
  }
}

11.2 数据隐私：PII 过滤、记忆隔离、删除权

Agent 处理的对话天然携带 PII（身份证号、手机号、住址、银行卡），而记忆系统又会让 PII 长期驻留。隐私工程的三道闸：

  1. 入口脱敏：PII 在进入 context 之前替换为占位符（如 PHONE_7f3a），Agent 全程只接触假名，真实值只存在受管控的映射表里。
  2. 记忆隔离：vector namespace / 表名按 tenant_id 分片；检索时强制带租户过滤条件——漏掉这一个条件就是一次数据泄露事故。
  3. 删除权：用户行使删除权时，要能删记忆向量、删会话日志、删审计里的明文片段（但保留操作记录的哈希链）。

合规基线：GDPR（欧盟）、中国《个人信息保护法》、HIPAA（美国医疗）。共性是三件事：告知同意、最小必要、可删除。

Python 版本

"""
PII 脱敏 + 租户隔离 —— Python
入口：mask() 之后内容才会进入 prompt。真实映射只在 Vault 内，Agent 永远看不到。
"""
from __future__ import annotations
import re, uuid
from dataclasses import dataclass, field

PATTERNS = {
    "PHONE":  r"1[3-9]\d{9}",
    "IDCARD": r"\d{17}[\dXx]",
    "EMAIL":  r"[\w.+-]+@[\w-]+\.[\w.]+",
}

@dataclass
class MaskVault:
    """假名映射表。进程重启即失效；落库版本应带 TTL 与访问审计。"""
    table: dict[str, str] = field(default_factory=dict)      # alias → real
    by_real: dict[str, str] = field(default_factory=dict)   # real → alias

    def mask(self, text: str) -> str:
        for kind, pat in PATTERNS.items():
            for real in re.findall(pat, text):
                if real not in self.by_real:
                    alias = f"{kind}_{uuid.uuid4().hex[:6]}"
                    self.by_real[real] = alias
                    self.table[alias] = real
                text = text.replace(real, self.by_real[real])
        return text

    def restore(self, text: str) -> str:
        """只在确有必要回填给用户时使用，且必须记审计。"""
        for alias, real in self.table.items():
            text = text.replace(alias, real)
        return text

def retrieve_memories(query_vec, tenant_id: str, store, top_k: int = 5):
    """记忆检索强制注入租户过滤——防跨租户串数据的最后一道闸。"""
    if not tenant_id:
        raise ValueError("tenant_id 必填，禁止无租户检索")
    return store.search(
        query_vec, top_k=top_k,
        filter={"tenant_id": {"$eq": tenant_id}},   # 漏这行 = 泄露事故
    )

vault = MaskVault()
raw = "客户张三 13812345678，身份证 110101199003077758，邮箱 zhang@example.com"
masked = vault.mask(raw)
print(masked)                    # 客户张三 PHONE_xxxx，身份证 IDCARD_xxxx，邮箱 EMAIL_xxxx
print(vault.restore(masked))     # 仅在确有必要回填给用户时使用

TypeScript 版本

/**
 * PII 脱敏 + 租户隔离 —— TypeScript
 */
import { randomBytes } from "node:crypto";

const PATTERNS: Array<[string, RegExp]> = [
  ["PHONE",  /1[3-9]\d{9}/g],
  ["IDCARD", /\d{17}[\dXx]/g],
  ["EMAIL",  /[\w.+-]+@[\w-]+\.[\w.]+/g],
];

export class MaskVault {
  private table = new Map<string, string>();   // alias → real
  private byReal = new Map<string, string>();   // real → alias

  mask(text: string): string {
    for (const [kind, pat] of PATTERNS) {
      for (const real of text.match(pat) ?? []) {
        if (!this.byReal.has(real)) {
          const alias = `${kind}_${randomBytes(3).toString("hex")}`;
          this.byReal.set(real, alias);
          this.table.set(alias, real);
        }
        text = text.replaceAll(real, this.byReal.get(real)!);
      }
    }
    return text;
  }

  restore(text: string): string {
    for (const [alias, real] of this.table) text = text.replaceAll(alias, real);
    return text;
  }
}

/** 记忆检索：租户过滤是硬约束，缺 tenant_id 直接抛错 */
export function tenantFilter(tenantId: string) {
  if (!tenantId) throw new Error("tenant_id 必填，禁止无租户检索");
  return { tenant_id: { $eq: tenantId } };
}

11.3 内容安全：多层防御与审计不可抵赖

Agent 被攻击的路径比传统应用多：prompt 注入（用户诱导 Agent 越权）、工具滥用（让 Agent 反复调用计费接口）、输出投毒（把恶意指令写进返回内容再让别的 Agent 执行）。防御按四层部署：

层手段拦住什么成本
输入层敏感词 / 注入特征分类器明显的注入与越权请求低
模型层安全对齐模型 + system prompt 约束 + 低温度大部分社会工程攻击中
输出层分类器二审 + 关键词兜底有害输出、越权承诺中
行为层工具网关、速率限制、异常模式检测工具滥用、批量攻击中

审计不可抵赖的关键：日志用「追加写 + 哈希链」（见 11.1 的 AuditLog），篡改任意一条都会导致后续哈希对不上。审计日志至少要能回答三个问题：谁（哪个会话）在什么时间调用了什么工具、参数是什么、结果是否被放行。

11.4 合规上线：企业内部审批流程

Agent 不是普通功能上线，按「高危变更」走流程：

  1. 内部试用（1–2 周）：研发 + 运营日常使用，收集工具误调用率、越权尝试次数。
  2. 安全评审：安全团队审工具权限表、数据流图、PII 处理、审计方案；高危项不过，不上线。
  3. 灰度发布：按会话百分比（5% → 20% → 100%）放量，每档观察 48 小时错误率与预算消耗。
  4. 用户告知：首轮对话明示「你在与 AI 助手交互，哪些数据会被记录、保留多久」。
  5. 回滚预案：保留一键降级开关——Agent 不可用时，业务能退回原人工 / 规则流程。

本章要点

  • 安全不是评审出来的，是运行时每一层编译出来的：网关、脱敏、审核、审计。
  • 权限最小化的落点：工具注册白名单 + 危险操作人工确认 + 配额与限额。
  • 隐私三闸：入口脱敏、租户隔离检索、可删除（含记忆向量）。
  • 审计用追加写哈希链，让「谁在什么时间对 Agent 做了什么」不可抵赖。
