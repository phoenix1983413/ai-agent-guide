第 12 章

Agent 的自我进化

RLHF / 经验蒸馏 / 记忆库整理 / 演化边界

学习目标

  • 区分 RLHF、经验蒸馏、知识蒸馏三条进化路线的适用场景
  • 设计「Agent 可自己改」与「必须人工评审」的边界
  • 为冻结参数的知识建立保鲜机制

章节地图：12.1 RLHF → 12.2 经验蒸馏 → 12.3 记忆整理 → 12.4 演化边界 → 12.5 持续学习与知识更新 → 本章产出。

12.1 从反馈中学习：RLHF 在 Agent 中的应用

RLHF（人类反馈强化学习）在 Agent 中的落地比在 Chatbot 中更复杂，因为 Agent 的动作空间是工具调用序列，不是单一文本。两种适配思路：

  • outcome-based RLHF：只对最终结果打分（成功 / 失败），用奖励模型训练策略。
  • process-based RLHF：对每一步（思考、工具调用）都打分，奖励高质量中间步骤。

实践建议：初期用 outcome-based，实现简单；数据量足够后再切 process-based，效果更好但标注成本高。

12.2 经验蒸馏：把多次执行压缩成可复用的策略

Agent 运行时会积累大量 trace。经验蒸馏是从这些 trace 中提炼出可复用的策略：

  1. 收集成功 trace，按任务类型聚类。
  2. 提取高频工具调用序列作为「脚本」。
  3. 把脚本编码成 prompt 片段或 decision tree，下次遇到同类任务直接命中。
  4. 定期用新 trace 刷新策略，避免过时。

12.3 记忆库的自动整理与知识蒸馏

记忆库会越积越多，需要自动整理：

  • 去重：相似度 > 0.95 的记忆合并。
  • 降级：长期未被访问的记忆从向量库移到冷存储。
  • 蒸馏：把多条相关记忆总结成一条高密度摘要，释放空间。

12.4 演化边界：什么该让 Agent 自己改，什么必须人工评审

层级Agent 可自主修改必须人工评审
提示词变量、示例、温度system prompt 核心规则、安全约束
工具参数默认值、重试策略新增工具、权限变更、删除接口
记忆自动摘要、冷热分层用户数据删除、跨租户合并
模型路由策略、fallback 顺序底层模型替换、安全对齐开关

12.5 持续学习与知识更新

模型参数是冻结的，世界却在持续变化。Agent 的「持续学习」分两条腿：经验腿（前几节的蒸馏与反思）和知识腿（本节）。知识腿解决「新事件、新数据、新政策进来后，Agent 怎么知道」。

更新方式时效成本适用
检索式更新分钟级（入库即可用）低绝大多数事实性知识：文档、公告、价格
提示词更新小时级（需回归测试）低行为规则、话术、流程变化
微调更新天级（数据 + 训练 + 评测）高稳定的风格与格式偏好，不存易变事实

三条铁律：① 易变事实永远走检索，不走微调——微调进去的知识无法低成本删除， model 会固执地「记得」过期的价格；② 知识要带生效日期，检索时按时间窗口过滤，防止 Agent 引用去年政策回答今年问题；③ 更新链路本身要评测——新知识入库后，用「新旧知识冲突」的问题做回归，确保 Agent 用新的不用旧的。

数据飞轮的伦理边界：用户交互数据用于改进系统时，要走 11.2 节的脱敏与授权；「用用户数据炼出的能力服务其他用户」必须有明确协议，这不是技术问题是信任问题。

动手实验

  1. 列出你的 Agent 依赖的 TOP10 易变事实（价格、政策、联系人），逐个标注更新方式和负责人。
  2. 给你的知识库加「生效日期」字段，构造三条「新旧知识冲突」的回归问题。
  3. 检查经验库（12.2）里的教训：哪几条其实已经过时了？给经验也加上保质期。

本章产出：带反馈闭环的 Agent 自迭代框架（双语言）

Python 版本

"""
Agent 自迭代框架 —— Python
核心：收集反馈 -> 评分 -> 蒸馏策略 -> 更新 prompt
"""
from __future__ import annotations
import json, hashlib
from dataclasses import dataclass, field
from typing import Any
from collections import defaultdict

@dataclass
class Trace:
    id: str
    task_type: str
    input: str
    steps: list[dict]
    outcome: str  # success | failure
    feedback_score: float = 0.0  # 0~1

class ExperienceDistiller:
    def __init__(self):
        self.traces: list[Trace] = []
        self.strategies: dict[str, list[dict]] = defaultdict(list)

    def add_trace(self, trace: Trace):
        self.traces.append(trace)

    def distill(self) -> dict[str, list[dict]]:
        """按任务类型聚类，提炼高频步骤序列作为策略。"""
        groups: dict[str, list[Trace]] = defaultdict(list)
        for t in self.traces:
            if t.outcome == "success":
                groups[t.task_type].append(t)

        for task_type, traces in groups.items():
            step_counter: dict[tuple, int] = defaultdict(int)
            for trace in traces:
                seq = tuple(s.get("tool") or s.get("action") for s in trace.steps)
                step_counter[seq] += 1
            # 取 Top-3 高频序列
            top = sorted(step_counter.items(), key=lambda x: x[1], reverse=True)[:3]
            self.strategies[task_type] = [
                {"sequence": list(seq), "count": count, "confidence": count / len(traces)}
                for seq, count in top
            ]
        return dict(self.strategies)

    def update_prompt(self, task_type: str) -> str:
        """根据蒸馏出的策略生成 prompt 片段。"""
        strats = self.strategies.get(task_type, [])
        if not strats:
            return ""
        lines = [f"任务类型: {task_type} 的推荐执行步骤："]
        for i, s in enumerate(strats, 1):
            lines.append(f"{i}. " + " -> ".join(s["sequence"]) + f" (置信度 {s['confidence']:.0%})")
        return "\n".join(lines)

# 使用示例
if __name__ == "__main__":
    distiller = ExperienceDistiller()
    distiller.add_trace(Trace(
        id="t1", task_type="search_and_summarize",
        input="AI Agent 最新进展",
        steps=[{"tool": "search", "query": "AI Agent 2024"}, {"tool": "summarize"}],
        outcome="success", feedback_score=0.9,
    ))
    distiller.add_trace(Trace(
        id="t2", task_type="search_and_summarize",
        input="AI Agent 最新进展",
        steps=[{"tool": "search", "query": "AI Agent 2024"}, {"tool": "summarize"}],
        outcome="success", feedback_score=0.85,
    ))
    print(distiller.update_prompt("search_and_summarize"))

TypeScript 版本

/**
Agent 自迭代框架 —— TypeScript
*/
interface Trace {
  id: string;
  taskType: string;
  input: string;
  steps: Array>;
  outcome: "success" | "failure";
  feedbackScore: number;
}

class ExperienceDistiller {
  private traces: Trace[] = [];
  private strategies = new Map>();

  addTrace(trace: Trace) { this.traces.push(trace); }

  distill(): Record> {
    const groups = new Map();
    for (const t of this.traces) {
      if (t.outcome !== "success") continue;
      const arr = groups.get(t.taskType) ?? [];
      arr.push(t);
      groups.set(t.taskType, arr);
    }

    for (const [taskType, traces] of groups) {
      const counter = new Map();
      for (const trace of traces) {
        const seq = trace.steps.map(s => s.tool ?? s.action ?? "").join("->");
        counter.set(seq, (counter.get(seq) ?? 0) + 1);
      }
      const top = Array.from(counter.entries())
        .sort((a, b) => b[1] - a[1])
        .slice(0, 3)
        .map(([sequence, count]) => ({
          sequence: sequence.split("->"),
          count,
          confidence: count / traces.length,
        }));
      this.strategies.set(taskType, top);
    }
    return Object.fromEntries(this.strategies);
  }

  updatePrompt(taskType: string): string {
    const strats = this.strategies.get(taskType);
    if (!strats?.length) return "";
    const lines = [`任务类型: ${taskType} 的推荐执行步骤：`];
    strats.forEach((s, i) => {
      lines.push(`${i + 1}. ${s.sequence.join(" -> ")} (置信度 ${(s.confidence * 100).toFixed(0)}%)`);
    });
    return lines.join("\n");
  }
}

  steps: [{ tool: "search", query: "AI Agent 2024" }, { tool: "summarize" }], outcome: "success", feedbackScore: 0.9 });
distiller.addTrace({ id: "t2", taskType: "search_and_summarize", input: "AI Agent 最新进展",
  steps: [{ tool: "search", query: "AI Agent 2024" }, { tool: "summarize" }], outcome: "success", feedbackScore: 0.85 });
