I/O 契约 / Chat UI / 命令面板 / 渐进式披露 / 信任设计

7.1 Agent 的 I/O 契约

Agent 与用户的交互不是简单的「一问一答」，而是持续的双向对话。设计 I/O 契约时要考虑三种形态：

  • 流式输出（Streaming）：Agent 的思考过程和最终答案同时推送，用户看到「正在思考…」→「逐步写出答案」。降低感知延迟。
  • 结构化事件（Structured Events）：不只是文本，还包括「工具调用中」「已搜索 3 条结果」「正在写文件」这类状态事件。前端据此渲染进度条、步骤条。
  • 异步通知（Async Notification）：Agent 在后台跑长任务，完成后主动推送（WebSocket / Web Push / 邮件）。用户无需盯着屏幕等。

好的 I/O 契约让 Agent 像一个「有反馈的同事」：你知道它在做什么、做到哪了、卡在哪。

7.3 渐进式披露

Agent 的执行过程可能很长，一次性展示所有中间步骤会让用户 overwhelm。渐进式披露的原则：

  1. 默认只展示最终答案和关键步骤：如「搜索了 3 个来源 → 综合得出…」。
  2. 细节默认折叠：用户点「展开」才看完整的工具调用参数和原始返回。
  3. 按用户角色调整 granularity：普通用户看结论，开发者看日志，管理员看审计。
  4. 异常时自动展开：工具报错、权限不足时，自动把错误详情和修复建议展示出来。

7.4 信任设计

用户信任 Agent 的前提是「知道它在干嘛」和「能随时叫停」。三个关键机制：

  • 来源标注：Agent 引用了外部信息时，显示来源链接或文件名。如「根据 [文件 A] 第 3 段…」。
  • 置信度显示：对不确定的答案标注置信度（高/中/低），或直接给出概率值。
  • 撤销机制：Agent 执行了不可逆操作（删除、发送、支付）后，提供「撤销」按钮，且窗口期不少于 5 秒。

下面是一个最小化的 WebSocket 聊天服务端 + 客户端，演示流式输出和结构化事件。

Python 服务端（FastAPI）

"""
WebSocket Agent 聊天原型 —— Python
依赖：fastapi, uvicorn, websockets
"""
from fastapi import FastAPI, WebSocket, WebSocketDisconnect
from fastapi.responses import HTMLResponse
import json, asyncio, uuid

app = FastAPI()

# ---- 连接管理 ----
class ConnectionManager:
    def __init__(self):
        self.active: dict[str, WebSocket] = {}

    async def connect(self, ws: WebSocket):
        cid = uuid.uuid4().hex[:8]
        await ws.accept()
        self.active[cid] = ws
        return cid

    def disconnect(self, cid: str):
        self.active.pop(cid, None)

    async def send(self, cid: str, event: dict):
        ws = self.active.get(cid)
        if ws:
            await ws.send_json(event)

manager = ConnectionManager()

# ---- 模拟 Agent 流式回复 ----
async def agent_stream(cid: str, user_msg: str):
    steps = [
        ("thinking", "正在理解问题…"),
        ("tool_call", "search", {"q": user_msg}),
        ("tool_result", "找到 3 条相关结果"),
        ("thinking", "正在综合答案…"),
        ("answer", f"关于「{user_msg}」的答案：这是一个模拟回复。"),
    ]
    for step in steps:
        await asyncio.sleep(0.6)  # 模拟耗时
        if step[0] == "thinking":
            await manager.send(cid, {"type": "thinking", "text": step[1]})
        elif step[0] == "tool_call":
            await manager.send(cid, {"type": "tool_call", "tool": step[1], "args": step[2]})
        elif step[0] == "tool_result":
            await manager.send(cid, {"type": "tool_result", "text": step[1]})
        elif step[0] == "answer":
            await manager.send(cid, {"type": "answer", "text": step[1]})

@app.websocket("/ws/chat")
async def ws_chat(ws: WebSocket):
    cid = await manager.connect(ws)
    try:
        while True:
            data = await ws.receive_json()
            user_msg = data.get("message", "")
            asyncio.create_task(agent_stream(cid, user_msg))
    except WebSocketDisconnect:
        manager.disconnect(cid)

@app.get("/")
async def get_client():
    return HTMLResponse("""

Agent Chat

  

  
  

""")

if __name__ == "__main__":
    import uvicorn
    uvicorn.run(app, host="0.0.0.0", port=8000)

TypeScript 服务端（Node.js + ws）

/**
WebSocket Agent 聊天原型 —— TypeScript
依赖：ws, @types/ws, typescript
*/
import { WebSocketServer, WebSocket } from "ws";

const wss = new WebSocketServer({ port: 8080 });
const clients = new Set();

wss.on("connection", (ws) => {
  clients.add(ws);
  ws.on("message", async (raw) => {
    const { message } = JSON.parse(raw.toString());
    const steps: Array<{ type: string; [k: string]: any }> = [
      { type: "thinking", text: "正在理解问题…" },
      { type: "tool_call", tool: "search", args: { q: message } },
      { type: "tool_result", text: "找到 3 条相关结果" },
      { type: "thinking", text: "正在综合答案…" },
      { type: "answer", text: `关于「${message}」的答案：这是一个模拟回复。` },
    ];
    for (const step of steps) {
      await new Promise(r => setTimeout(r, 600));
      ws.send(JSON.stringify(step));
    }
  });
  ws.on("close", () => clients.delete(ws));
});

console.log("Agent Chat WS 运行在 ws://localhost:8080");
