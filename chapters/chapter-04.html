第 4 章

工具使用与函数调用

工具声明、参数生成、执行观测、错误处理与安全边界

4.1 工具声明范式

Agent 的工具调用目前主流三种声明方式：

  • JSON Schema：OpenAI / Anthropic 原生支持，描述 name、description、parameters（JSON Schema）。最通用。
  • OpenAPI：适合已有 REST API 的场景，直接复用 Swagger / OpenAPI 文档生成工具声明。
  • 动态注册：运行时从数据库或配置中心加载工具列表，适合工具集经常变化的系统。

无论哪种范式，核心字段不变：name、description、parameters（含 required 和类型）。

4.2 参数生成：从自然语言到结构化调用

LLM 本身输出的是文本，不是 JSON。工具调用的可靠性依赖两点：

  • 提示词里的格式约束：明确告诉模型「只输出 JSON，不要多余的话」。
  • 函数调用模式（Function Calling）：OpenAI / Anthropic / Gemini 都原生支持，让模型直接输出结构化的 tool_call，比让模型自己写 JSON 可靠得多。

4.3 执行与观测

工具调用的完整生命周期：

  1. Agent 输出 tool_call（含 name + arguments）。
  2. 系统执行工具，捕获 stdout / 返回值 / 异常。
  3. 工具结果作为 role: tool 的消息回填到对话历史。
  4. Agent 继续下一步思考或行动。

关键：工具输出必须原样回填，不能由 Agent 「转述」，否则会引入幻觉。

4.4 工具错误处理

错误类型处理策略示例
瞬态错误指数退避重试（最多 3 次）网络超时、限流 429
永久错误立即停止，报告给用户参数校验失败、权限不足
业务错误把错误信息回填，让 Agent 自行决定下一步查询无结果、余额不足
未知错误兜底降级：返回统一错误消息 + 建议人工介入未预期的异常

4.5 安全边界

  • 最小权限：每个工具只给完成其功能所必需的最小权限。
  • 审计日志：每次工具调用记录 who / when / what / result。
  • 沙箱隔离：代码执行类工具必须在隔离环境（Docker / Firecracker）中运行。
  • 人工兜底：高风险操作（删除、发送、支付）强制人工确认。

以下示例注册 5 个常用工具：搜索、计算、文件读写、HTTP 请求、代码执行。

Python 版本

"""
5 个工具注册表示例 —— Python
依赖：openai >= 1.0, requests, aiofiles（异步文件）
"""
from __future__ import annotations
import json, os, re
from dataclasses import dataclass
from typing import Any
import openai, requests

client = openai.OpenAI()

# ---- 工具定义 ----
TOOLS = [
    {
        "type": "function",
        "function": {
            "name": "search",
            "description": "搜索互联网，返回摘要",
            "parameters": {
                "type": "object",
                "properties": {"q": {"type": "string", "description": "搜索关键词"}},
                "required": ["q"],
            },
        },
    },
    {
        "type": "function",
        "function": {
            "name": "calc",
            "description": "执行安全四则运算表达式",
            "parameters": {
                "type": "object",
                "properties": {"expr": {"type": "string", "description": "如 (3+4)*5"}},
                "required": ["expr"],
            },
        },
    },
    {
        "type": "function",
        "function": {
            "name": "read_file",
            "description": "读取指定路径的文本文件（白名单目录内）",
            "parameters": {
                "type": "object",
                "properties": {"path": {"type": "string"}},
                "required": ["path"],
            },
        },
    },
    {
        "type": "function",
        "function": {
            "name": "http_get",
            "description": "发送 HTTP GET 请求并返回前 500 字",
            "parameters": {
                "type": "object",
                "properties": {"url": {"type": "string", "format": "uri"}},
                "required": ["url"],
            },
        },
    },
    {
        "type": "function",
        "function": {
            "name": "run_code",
            "description": "在沙箱中执行 Python 代码（仅允许纯计算）",
            "parameters": {
                "type": "object",
                "properties": {"code": {"type": "string"}},
                "required": ["code"],
            },
        },
    },
]

# ---- 工具实现 ----
SAFE_DIR = os.path.abspath("./sandbox")

def tool_search(q: str) -> str:
    # 实际场景接搜索 API；这里返回模拟结果
    return f"[搜索摘要] 关于「{q}」的前三条结果……"

def tool_calc(expr: str) -> str:
    if not re.match(r"^[\d\s\+\-\*/\(\)\.]+$", expr):
        return "ERROR: 表达式包含非法字符"
    try:
        return str(eval(expr, {"__builtins__": {}}, {}))
    except Exception as e:
        return f"ERROR: {e}"

def tool_read_file(path: str) -> str:
    full = os.path.abspath(os.path.join(SAFE_DIR, path))
    if not full.startswith(SAFE_DIR):
        return "ERROR: 路径越权"
    if not os.path.exists(full):
        return "ERROR: 文件不存在"
    with open(full, "r", encoding="utf-8") as f:
        return f.read(2000)  # 限长

def tool_http_get(url: str) -> str:
    try:
        r = requests.get(url, timeout=10)
        return r.text[:500]
    except Exception as e:
        return f"ERROR: {e}"

def tool_run_code(code: str) -> str:
    # 生产环境必须用 Docker / Firecracker 隔离
    import subprocess, tempfile
    with tempfile.NamedTemporaryFile("w", suffix=".py", delete=False) as f:
        f.write(code)
        path = f.name
    try:
        result = subprocess.run(
            ["python3", path], capture_output=True, text=True, timeout=5
        )
        return result.stdout or result.stderr or "（无输出）"
    except subprocess.TimeoutExpired:
        return "ERROR: 执行超时"
    finally:
        os.unlink(path)

TOOL_IMPL = {
    "search": tool_search,
    "calc": tool_calc,
    "read_file": tool_read_file,
    "http_get": tool_http_get,
    "run_code": tool_run_code,
}

def call_tool(name: str, arguments: dict) -> str:
    fn = TOOL_IMPL.get(name)
    return fn(**arguments) if fn else f"ERROR: 未知工具 {name}"

# 使用示例
if __name__ == "__main__":
    print(call_tool("calc", {"expr": "(3+4)*5"}))
    print(call_tool("search", {"q": "AI Agent"}))

/**
 * 5 个工具注册表示例 —— TypeScript
 * 依赖：openai >= 4.0, axios
 */
import OpenAI from "openai";
import axios from "axios";
import { readFile, access, constants } from "node:fs/promises";
import { join } from "node:path";

const client = new OpenAI();

const TOOLS = [
  { type: "function" as const, function: {
      name: "search", description: "搜索互联网，返回摘要",
      parameters: { type: "object", properties: { q: { type: "string", description: "搜索关键词" } }, required: ["q"] } } },
  { type: "function" as const, function: {
      name: "calc", description: "执行安全四则运算表达式",
      parameters: { type: "object", properties: { expr: { type: "string", description: "如 (3+4)*5" } }, required: ["expr"] } } },
  { type: "function" as const, function: {
      name: "read_file", description: "读取白名单目录内的文本文件",
      parameters: { type: "object", properties: { path: { type: "string" } }, required: ["path"] } } },
  { type: "function" as const, function: {
      name: "http_get", description: "发送 HTTP GET 请求并返回前 500 字",
      parameters: { type: "object", properties: { url: { type: "string", format: "uri" } }, required: ["url"] } } },
  { type: "function" as const, function: {
      name: "run_code", description: "在沙箱中执行 Python 代码（仅允许纯计算）",
      parameters: { type: "object", properties: { code: { type: "string" } }, required: ["code"] } } },
];

const SAFE_DIR = join(process.cwd(), "sandbox");

async function callTool(name: string, args: Record): Promise {
  switch (name) {
    case "search":
      return `[搜索摘要] 关于「${args.q}」的前三条结果……`;
    case "calc": {
      if (!/^[\d\s+\-*/().]+$/.test(args.expr)) return "ERROR: 表达式包含非法字符";
      try { return String(eval(args.expr)); } catch (e: any) { return `ERROR: ${e.message}`; }
    }
    case "read_file": {
      const full = join(SAFE_DIR, args.path);
      try { await access(full, constants.R_OK); return (await readFile(full, "utf-8")).slice(0, 2000); }
      catch (e: any) { return `ERROR: ${e.code === "ENOENT" ? "文件不存在" : "路径越权"}`; }
    }
    case "http_get": {
      try { const r = await axios.get(args.url, { timeout: 10000 }); return r.data.slice(0, 500); }
      catch (e: any) { return `ERROR: ${e.message}`; }
    }
    case "run_code": {
      // 生产环境必须用 Docker / Firecracker 隔离
      const { spawn } = await import("node:child_process");
      const py = spawn("python3", ["-c", args.code]);
      let out = "", err = "";
      py.stdout.on("data", d => out += d);
      py.stderr.on("data", d => err += d);
      await new Promise((res, rej) => py.on("close", code => code === 0 ? res() : rej()));
      return out || err || "（无输出）";
    }
    default:
      return `ERROR: 未知工具 ${name}`;
  }
}

// 使用示例
(async () => {
  console.log(await callTool("calc", { expr: "(3+4)*5" }));
  console.log(await callTool("search", { q: "AI Agent" }));
})();
